import { formatComponentName, getFigmaComponentsData } from '@libs/figma'
import { getCollection } from 'astro:content'

export const allFigma = await getCollection('figma')

<div class="components-grid">
  {
    (await getFigmaComponentsData()).map(({ name: component, tabs }) => {
      const firstEntry = allFigma.find((e) => e.id === `${component}/${tabs[0]}`)
      const description = firstEntry?.data?.description || `View ${tabs.join(', ')} for the ${formatComponentName(component).toLowerCase()} component.`
      
      return (
        <div class="card component-card">
          <div class="card-content">
            <h3 class="card-title">
              <a href={`/figma/docs/components/${component}/${tabs[0]}`}>{formatComponentName(component)}</a>
            </h3>
            <p class="card-body">
              {description}
            </p>
            <ul class="nav">
              {tabs.map((tab) => (
                <li class="nav-item"><a href={`/figma/docs/components/${component}/${tab}`} class="nav-link">
                  {tab.charAt(0).toUpperCase() + tab.slice(1)}
                </a></li>
              ))}
            </ul>
          </div>
        </div>
      )
    })
  }
</div>

<style is:inline> {`

  .components-grid {
    display: grid;
    grid-template-columns: repeat(auto-fit, minmax(320px, 1fr));
    gap: 1.5rem;
  }
`}
</style>
